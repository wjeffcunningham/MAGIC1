import { supabase } from "./supabase.js"; import { getLocalSession } from "./session.js"; const list = document.getElementById("pending-list"); const notAdmin = document.getElementById("not-admin"); async function main() { const session = getLocalSession(); if (!session?.isAdmin) { notAdmin.classList.remove("hidden"); return; } loadPending(); } async function loadPending() { const { data, error } = await supabase .from("players") .select("*") .eq("status", "pending") .order("created_at", { ascending: true }); if (error) { list.innerHTML = `

Error loading players.

`; return; } if (!data.length) { list.innerHTML = `

No pending players.

`; list.classList.remove("hidden"); return; } list.innerHTML = ""; list.classList.remove("hidden"); data.forEach(player => { const card = document.createElement("div"); card.className = "bg-white p-5 rounded-xl shadow"; card.innerHTML = `

${player.full_name}

${player.email}

`; list.appendChild(card); }); // attach actions document.querySelectorAll(".approve").forEach(btn => btn.addEventListener("click", approvePlayer) ); document.querySelectorAll(".deny").forEach(btn => btn.addEventListener("click", denyPlayer) ); } async function approvePlayer(e) { const id = e.target.dataset.id; await supabase .from("players") .update({ status: "active" }) .eq("id", id); loadPending(); } async function denyPlayer(e) { const id = e.target.dataset.id; await supabase .from("players") .delete() .eq("id", id); loadPending(); } main();
